# Install order on a fresh machine (details in README "Set up from scratch"):
#
#   conda create -n diffdynamic python=3.8 -y
#   conda activate diffdynamic
#   pip install torch==2.4.1 --index-url https://download.pytorch.org/whl/cu118
#   pip install torch_geometric==2.6.1
#   pip install torch_scatter==2.1.2 torch_sparse==0.6.18 \
#       -f https://data.pyg.org/whl/torch-2.4.1+cu118.html
#   conda install -c conda-forge openbabel=3.1.0 -y
#   pip install -r requirements.txt
#   python -m pip install --no-deps --no-build-isolation -e .
#
# torch / torch_geometric / torch_scatter / torch_sparse are NOT listed below on
# purpose: pip would resolve them against the wrong PyTorch/CUDA build. The GPU
